import com.sun.net.httpserver.HttpExchange; import com.sun.net.httpserver.HttpServer; import java.io.IOException; import java.io.OutputStream; import java.net.InetSocketAddress; import java.net.URLDecoder; import java.nio.charset.StandardCharsets; import java.sql.*; public class JobTrackServer { private static final int PORT = 8080; // ========================================================= // START SERVER // ========================================================= public static void main(String[] args) throws Exception { HttpServer server = HttpServer.create( new InetSocketAddress(PORT), 0 ); // Main website server.createContext("/", exchange -> { serveFrontend(exchange); }); // Add application server.createContext("/add", exchange -> { addApplication(exchange); }); // View applications server.createContext("/applications", exchange -> { getApplications(exchange); }); // Delete application server.createContext("/delete", exchange -> { deleteApplication(exchange); }); // Update application server.createContext("/update", exchange -> { updateApplication(exchange); }); // View history server.createContext("/history", exchange -> { getHistory(exchange); }); server.start(); System.out.println( "JobTrack Server started!" ); System.out.println( "Open: http://localhost:8080/" ); } // ========================================================= // SERVE FRONTEND // ========================================================= private static void serveFrontend( HttpExchange exchange) throws IOException { try { java.nio.file.Path path = java.nio.file.Paths.get( "frontend", "index.html" ); String html = java.nio.file.Files.readString( path ); sendHtmlResponse( exchange, html ); } catch (Exception e) { e.printStackTrace(); sendResponse( exchange, "Unable to load JobTrack website." ); } } // ========================================================= // ADD APPLICATION // ========================================================= private static void addApplication( HttpExchange exchange) throws IOException { addCors(exchange); String query = exchange.getRequestURI().getQuery(); String company = getValue(query, "company"); String role = getValue(query, "role"); String jobReferenceId = getValue( query, "jobReferenceId" ); String status = getValue(query, "status"); String location = getValue(query, "location"); String jobLink = getValue(query, "jobLink"); String notes = getValue(query, "notes"); String applicationDate = java.time.LocalDate .now() .toString(); try (Connection conn = Database.getConnection()) { // ================================================= // GENERATE JOBTRACK ID // ================================================= String jobTrackId; String idSql = "SELECT COALESCE(MAX(id), 0) + 1 " + "FROM applications"; try ( PreparedStatement idStmt = conn.prepareStatement(idSql); ResultSet idRs = idStmt.executeQuery() ) { idRs.next(); int nextId = idRs.getInt(1); jobTrackId = String.format( "JT-%04d", nextId ); } // ================================================= // INSERT APPLICATION // ================================================= String sql = """ INSERT INTO applications ( jobtrack_id, job_reference_id, company, role, application_date, status, location, job_link, notes ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?) """; try ( PreparedStatement stmt = conn.prepareStatement( sql, Statement.RETURN_GENERATED_KEYS ) ) { stmt.setString( 1, jobTrackId ); stmt.setString( 2, jobReferenceId ); stmt.setString( 3, company ); stmt.setString( 4, role ); stmt.setString( 5, applicationDate ); stmt.setString( 6, status ); stmt.setString( 7, location ); stmt.setString( 8, jobLink ); stmt.setString( 9, notes ); stmt.executeUpdate(); int applicationId = -1; try ( ResultSet keys = stmt.getGeneratedKeys() ) { if (keys.next()) { applicationId = keys.getInt(1); } } // ================================================= // ADD INITIAL HISTORY // ================================================= if ( applicationId != -1 && !status.isEmpty() ) { String historySql = """ INSERT INTO application_history ( application_id, status, changed_at ) VALUES (?, ?, ?) """; try ( PreparedStatement historyStmt = conn.prepareStatement( historySql ) ) { historyStmt.setInt( 1, applicationId ); historyStmt.setString( 2, status ); historyStmt.setString( 3, java.time.LocalDateTime .now() .toString() ); historyStmt.executeUpdate(); } } sendResponse( exchange, "Application added successfully!\n\n" + "JobTrack ID: " + jobTrackId ); } } catch (SQLException e) { e.printStackTrace(); sendResponse( exchange, "Error adding application: " + e.getMessage() ); } } // ========================================================= // GET APPLICATIONS // ========================================================= private static void getApplications( HttpExchange exchange) throws IOException { addCors(exchange); String query = exchange.getRequestURI().getQuery(); String search = getValue(query, "search"); String statusFilter = getValue(query, "status"); if (statusFilter.isEmpty()) { statusFilter = "All"; } String sql = """ SELECT * FROM applications WHERE ( company LIKE ? OR role LIKE ? OR jobtrack_id LIKE ? OR job_reference_id LIKE ? ) AND ( ? = 'All' OR status = ? ) ORDER BY id DESC """; StringBuilder applicationsHtml = new StringBuilder(); int count = 0; try ( Connection conn = Database.getConnection(); PreparedStatement stmt = conn.prepareStatement(sql) ) { String searchPattern = "%" + search + "%"; stmt.setString( 1, searchPattern ); stmt.setString( 2, searchPattern ); stmt.setString( 3, searchPattern ); stmt.setString( 4, searchPattern ); stmt.setString( 5, statusFilter ); stmt.setString( 6, statusFilter ); ResultSet rs = stmt.executeQuery(); while (rs.next()) { count++; int id = rs.getInt("id"); String jobTrackId = escapeHtml( rs.getString( "jobtrack_id" ) ); String jobReferenceId = escapeHtml( rs.getString( "job_reference_id" ) ); String company = escapeHtml( rs.getString( "company" ) ); String role = escapeHtml( rs.getString( "role" ) ); String date = escapeHtml( rs.getString( "application_date" ) ); String status = escapeHtml( rs.getString( "status" ) ); String location = escapeHtml( rs.getString( "location" ) ); String jobLink = escapeHtml( rs.getString( "job_link" ) ); String notes = escapeHtml( rs.getString( "notes" ) ); applicationsHtml.append("""
%s

%s

""".formatted( company, role, status.toLowerCase(), status, jobTrackId, jobReferenceId.isEmpty() ? "" : """
Job ID / Registration ID: %s
""".formatted( jobReferenceId ), date, location )); if (!jobLink.isEmpty()) { applicationsHtml.append("""
Open Job Posting
""".formatted( jobLink )); } if (!notes.isEmpty()) { applicationsHtml.append("""
Notes: %s
""".formatted( notes )); } applicationsHtml.append("""
""".formatted( id, id, id )); } } catch (SQLException e) { e.printStackTrace(); sendResponse( exchange, "Database error: " + e.getMessage() ); return; } String html = """ JobTrack - Applications

JobTrack

Manage and track your job applications

My Applications

%d Application%s

%s
""".formatted( count, count == 1 ? "" : "s", escapeHtml(search), statusFilter.equals("All") ? "selected" : "", statusFilter.equals("Applied") ? "selected" : "", statusFilter.equals("Shortlisted") ? "selected" : "", statusFilter.equals("Interview") ? "selected" : "", statusFilter.equals("Selected") ? "selected" : "", statusFilter.equals("Rejected") ? "selected" : "", count == 0 ? """
No applications found.
""" : applicationsHtml.toString() ); sendHtmlResponse( exchange, html ); } // ========================================================= // VIEW APPLICATION HISTORY // ========================================================= private static void getHistory( HttpExchange exchange) throws IOException { addCors(exchange); String query = exchange.getRequestURI().getQuery(); String idValue = getValue(query, "id"); StringBuilder historyHtml = new StringBuilder(); try ( Connection conn = Database.getConnection() ) { int applicationId = Integer.parseInt(idValue); String sql = """ SELECT status, changed_at FROM application_history WHERE application_id = ? ORDER BY id ASC """; try ( PreparedStatement stmt = conn.prepareStatement(sql) ) { stmt.setInt( 1, applicationId ); ResultSet rs = stmt.executeQuery(); while (rs.next()) { String status = escapeHtml( rs.getString( "status" ) ); String changedAt = escapeHtml( rs.getString( "changed_at" ) ); historyHtml.append("""
%s
%s
""".formatted( status, changedAt )); } } } catch (Exception e) { e.printStackTrace(); sendResponse( exchange, "Error loading history." ); return; } String html = """ JobTrack - Application History

JobTrack

Application Status History

Application History

%s
Back to Applications
""".formatted( historyHtml.length() == 0 ? """
No history available.
""" : historyHtml.toString() ); sendHtmlResponse( exchange, html ); } // ========================================================= // DELETE APPLICATION // ========================================================= private static void deleteApplication( HttpExchange exchange) throws IOException { addCors(exchange); String query = exchange.getRequestURI().getQuery(); String idValue = getValue(query, "id"); try ( Connection conn = Database.getConnection() ) { int id = Integer.parseInt(idValue); // Delete history first String historySql = """ DELETE FROM application_history WHERE application_id = ? """; try ( PreparedStatement historyStmt = conn.prepareStatement( historySql ) ) { historyStmt.setInt( 1, id ); historyStmt.executeUpdate(); } // Delete application String sql = "DELETE FROM applications WHERE id = ?"; try ( PreparedStatement stmt = conn.prepareStatement(sql) ) { stmt.setInt( 1, id ); int rows = stmt.executeUpdate(); if (rows > 0) { sendResponse( exchange, "Application deleted successfully!" ); } else { sendResponse( exchange, "Application not found." ); } } } catch (Exception e) { e.printStackTrace(); sendResponse( exchange, "Error deleting application." ); } } // ========================================================= // UPDATE APPLICATION // ========================================================= private static void updateApplication( HttpExchange exchange) throws IOException { addCors(exchange); String query = exchange.getRequestURI().getQuery(); String id = getValue(query, "id"); String company = getValue(query, "company"); String role = getValue(query, "role"); String jobReferenceId = getValue( query, "jobReferenceId" ); String status = getValue(query, "status"); String location = getValue(query, "location"); String jobLink = getValue(query, "jobLink"); String notes = getValue(query, "notes"); try ( Connection conn = Database.getConnection() ) { int applicationId = Integer.parseInt(id); // ================================================= // GET OLD STATUS // ================================================= String oldStatus = ""; String oldStatusSql = """ SELECT status FROM applications WHERE id = ? """; try ( PreparedStatement oldStatusStmt = conn.prepareStatement( oldStatusSql ) ) { oldStatusStmt.setInt( 1, applicationId ); try ( ResultSet rs = oldStatusStmt.executeQuery() ) { if (rs.next()) { oldStatus = rs.getString( "status" ); if (oldStatus == null) { oldStatus = ""; } } } } // ================================================= // UPDATE APPLICATION // ================================================= String sql = """ UPDATE applications SET company = ?, role = ?, job_reference_id = ?, status = ?, location = ?, job_link = ?, notes = ? WHERE id = ? """; try ( PreparedStatement stmt = conn.prepareStatement(sql) ) { stmt.setString( 1, company ); stmt.setString( 2, role ); stmt.setString( 3, jobReferenceId ); stmt.setString( 4, status ); stmt.setString( 5, location ); stmt.setString( 6, jobLink ); stmt.setString( 7, notes ); stmt.setInt( 8, applicationId ); int rows = stmt.executeUpdate(); if (rows > 0) { // ================================================= // ADD STATUS HISTORY ONLY IF STATUS CHANGED // ================================================= if ( !status.equals(oldStatus) && !status.isEmpty() ) { String historySql = """ INSERT INTO application_history ( application_id, status, changed_at ) VALUES (?, ?, ?) """; try ( PreparedStatement historyStmt = conn.prepareStatement( historySql ) ) { historyStmt.setInt( 1, applicationId ); historyStmt.setString( 2, status ); historyStmt.setString( 3, java.time.LocalDateTime .now() .toString() ); historyStmt.executeUpdate(); } } sendResponse( exchange, "Application updated successfully!" ); } else { sendResponse( exchange, "Application not found." ); } } } catch (Exception e) { e.printStackTrace(); sendResponse( exchange, "Error updating application." ); } } // ========================================================= // GET QUERY VALUE // ========================================================= private static String getValue( String query, String key) { if ( query == null || query.isEmpty() ) { return ""; } String[] pairs = query.split("&"); for (String pair : pairs) { String[] parts = pair.split( "=", 2 ); if (parts.length == 2) { String currentKey = URLDecoder.decode( parts[0], StandardCharsets.UTF_8 ); if ( currentKey.equals(key) ) { return URLDecoder.decode( parts[1], StandardCharsets.UTF_8 ); } } } return ""; } // ========================================================= // HTML ESCAPING // ========================================================= private static String escapeHtml( String text) { if (text == null) { return ""; } return text .replace( "&", "&" ) .replace( "<", "<" ) .replace( ">", ">" ) .replace( "\"", """ ) .replace( "'", "'" ); } // ========================================================= // CORS // ========================================================= private static void addCors( HttpExchange exchange) { exchange.getResponseHeaders() .add( "Access-Control-Allow-Origin", "*" ); exchange.getResponseHeaders() .add( "Access-Control-Allow-Methods", "GET, OPTIONS" ); exchange.getResponseHeaders() .add( "Access-Control-Allow-Headers", "Content-Type" ); } // ========================================================= // SEND TEXT RESPONSE // ========================================================= private static void sendResponse( HttpExchange exchange, String response) throws IOException { byte[] bytes = response.getBytes( StandardCharsets.UTF_8 ); exchange.getResponseHeaders() .set( "Content-Type", "text/plain; charset=UTF-8" ); exchange.sendResponseHeaders( 200, bytes.length ); try ( OutputStream os = exchange.getResponseBody() ) { os.write(bytes); } } // ========================================================= // SEND HTML RESPONSE // ========================================================= private static void sendHtmlResponse( HttpExchange exchange, String html) throws IOException { byte[] bytes = html.getBytes( StandardCharsets.UTF_8 ); exchange.getResponseHeaders() .set( "Content-Type", "text/html; charset=UTF-8" ); exchange.sendResponseHeaders( 200, bytes.length ); try ( OutputStream os = exchange.getResponseBody() ) { os.write(bytes); } } }
